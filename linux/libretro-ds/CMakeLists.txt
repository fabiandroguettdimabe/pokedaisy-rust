cmake_minimum_required(VERSION 3.18)
project(pokedaisy_ds_libretro LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(LR ${ROOT}/third_party/dsperate-lr)
set(DS ${LR}/DSperate)
if(NOT EXISTS ${DS}/src/core/nds.h)
  message(FATAL_ERROR "Run git submodule update --init --recursive third_party/dsperate-lr")
endif()
foreach(feature DSPERATE_SDL DSPERATE_HEADLESS DSPERATE_TESTS DSPERATE_CHEEVOS DSPERATE_NET DSPERATE_VULKAN)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${DS} dsperate EXCLUDE_FROM_ALL)
if(WIN32)
  set(DS_WINDOWS_COMPAT ${CMAKE_CURRENT_SOURCE_DIR}/windows_compat)
  set(DS_WINDOWS_OVERLAY ${CMAKE_CURRENT_BINARY_DIR}/windows_compat/include)
  set(DS_PAGE_TABLE_SOURCE ${DS}/src/core/mem/page_table.h)
  set(DS_PAGE_TABLE_OVERLAY ${DS_WINDOWS_OVERLAY}/core/mem/page_table.h)
  file(READ ${DS_PAGE_TABLE_SOURCE} DS_PAGE_TABLE_TEXT)
  string(FIND "${DS_PAGE_TABLE_TEXT}" "#include <cstdlib>" DS_CSTDLIB_INCLUDE)
  string(FIND "${DS_PAGE_TABLE_TEXT}" "std::aligned_alloc(PAGE_SIZE, n)" DS_ALIGNED_ALLOC_CALL)
  string(FIND "${DS_PAGE_TABLE_TEXT}" "if (owned) std::free(p);" DS_ALIGNED_FREE_CALL)
  if(DS_CSTDLIB_INCLUDE EQUAL -1 OR DS_ALIGNED_ALLOC_CALL EQUAL -1 OR DS_ALIGNED_FREE_CALL EQUAL -1)
    message(FATAL_ERROR "Pinned DSperate page_table.h no longer matches the Windows compatibility patch")
  endif()
  string(REPLACE "#include <cstdlib>" "#include <cstdlib>\n#include <windows_compat/aligned_alloc.h>" DS_PAGE_TABLE_TEXT "${DS_PAGE_TABLE_TEXT}")
  string(REPLACE "std::aligned_alloc(PAGE_SIZE, n)" "vyntra_ds_windows::aligned_alloc(PAGE_SIZE, n)" DS_PAGE_TABLE_TEXT "${DS_PAGE_TABLE_TEXT}")
  string(REPLACE "if (owned) std::free(p);" "if (owned) vyntra_ds_windows::aligned_free(p);" DS_PAGE_TABLE_TEXT "${DS_PAGE_TABLE_TEXT}")
  file(MAKE_DIRECTORY ${DS_WINDOWS_OVERLAY}/core/mem)
  file(WRITE ${DS_PAGE_TABLE_OVERLAY} "${DS_PAGE_TABLE_TEXT}")
  target_include_directories(dsperate_core BEFORE PRIVATE ${DS_WINDOWS_OVERLAY} ${DS_WINDOWS_COMPAT}/include)
  target_sources(dsperate_core PRIVATE ${DS_WINDOWS_COMPAT}/windows_mman.cpp)
  set(DS_ANALOG_SOURCE ${DS}/src/core/cheat/analog_db.cpp)
  set(DS_ANALOG_OVERLAY ${CMAKE_CURRENT_BINARY_DIR}/windows_compat/analog_db.cpp)
  file(READ ${DS_ANALOG_SOURCE} DS_ANALOG_TEXT)
  string(FIND "${DS_ANALOG_TEXT}" ".section .rodata.dsperate_analog_db" DS_ANALOG_ELF_SECTION)
  string(FIND "${DS_ANALOG_TEXT}" ".previous" DS_ANALOG_ELF_PREVIOUS)
  if(DS_ANALOG_ELF_SECTION EQUAL -1 OR DS_ANALOG_ELF_PREVIOUS EQUAL -1)
    message(FATAL_ERROR "Pinned DSperate analog_db.cpp no longer matches the Windows section patch")
  endif()
  string(REPLACE ".section .rodata.dsperate_analog_db" ".section .rdata" DS_ANALOG_TEXT "${DS_ANALOG_TEXT}")
  string(REPLACE [=[,\"a\",%progbits]=] [=[,\"dr\"]=] DS_ANALOG_TEXT "${DS_ANALOG_TEXT}")
  string(REPLACE ".previous" ".text" DS_ANALOG_TEXT "${DS_ANALOG_TEXT}")
  file(WRITE ${DS_ANALOG_OVERLAY} "${DS_ANALOG_TEXT}")
  get_target_property(DS_CORE_SOURCES dsperate_core SOURCES)
  list(FILTER DS_CORE_SOURCES EXCLUDE REGEX "(^|/)cheat/analog_db\\.cpp$")
  set_property(TARGET dsperate_core PROPERTY SOURCES "${DS_CORE_SOURCES}")
  target_sources(dsperate_core PRIVATE ${DS_ANALOG_OVERLAY})
  set_source_files_properties(${DS_ANALOG_OVERLAY} PROPERTIES
    COMPILE_DEFINITIONS "DSPERATE_ANALOG_DB_DIR=\"${DS}/src/core/cheat/analog_db\"")
  set_source_files_properties(${DS_ANALOG_OVERLAY} PROPERTIES OBJECT_DEPENDS
    ${DS}/src/core/cheat/analog_db/usrcheat-analog.dat)
  set(DS_FONT_SOURCE ${DS}/src/core/io/dsi_font.cpp)
  set(DS_FONT_OVERLAY ${CMAKE_CURRENT_BINARY_DIR}/windows_compat/dsi_font.cpp)
  file(READ ${DS_FONT_SOURCE} DS_FONT_TEXT)
  string(FIND "${DS_FONT_TEXT}" ".section .rodata.dsperate_dsi_font" DS_FONT_SECTION)
  string(FIND "${DS_FONT_TEXT}" ".previous" DS_FONT_PREVIOUS)
  if(DS_FONT_SECTION EQUAL -1 OR DS_FONT_PREVIOUS EQUAL -1)
    message(FATAL_ERROR "Pinned DSperate dsi_font.cpp no longer matches the Windows section patch")
  endif()
  string(REPLACE ".section .rodata.dsperate_dsi_font" ".section .rdata" DS_FONT_TEXT "${DS_FONT_TEXT}")
  string(REPLACE [=[,\"a\",%progbits]=] [=[,\"dr\"]=] DS_FONT_TEXT "${DS_FONT_TEXT}")
  string(REPLACE ".previous" ".text" DS_FONT_TEXT "${DS_FONT_TEXT}")
  file(WRITE ${DS_FONT_OVERLAY} "${DS_FONT_TEXT}")
  get_target_property(DS_CORE_SOURCES dsperate_core SOURCES)
  list(FILTER DS_CORE_SOURCES EXCLUDE REGEX "(^|/)io/dsi_font\\.cpp$")
  set_property(TARGET dsperate_core PROPERTY SOURCES "${DS_CORE_SOURCES}")
  target_sources(dsperate_core PRIVATE ${DS_FONT_OVERLAY})
  set_source_files_properties(${DS_FONT_OVERLAY} PROPERTIES
    COMPILE_DEFINITIONS "DSPERATE_DSI_FONT_DIR=\"${DS}/src/core/io/dsi_font\"")
  set_source_files_properties(${DS_FONT_OVERLAY} PROPERTIES OBJECT_DEPENDS
    "${DS}/src/core/io/dsi_font/TWLFontTable.dat;${DS}/src/core/io/dsi_font/TWLFontTable-cn.dat;${DS}/src/core/io/dsi_font/TWLFontTable-kr.dat")
  target_compile_definitions(dsperate_core PRIVATE
    DSPERATE_ANALOG_DB_DIR=\"${DS}/src/core/cheat/analog_db\"
    DSPERATE_DSI_FONT_DIR=\"${DS}/src/core/io/dsi_font\")
  set_source_files_properties(
    ${DS}/src/core/cart/rom_source.cpp
    ${DS}/src/core/cart/zip_cache.cpp
    TARGET_DIRECTORY dsperate_core
    PROPERTIES COMPILE_OPTIONS "-include;${DS_WINDOWS_COMPAT}/posix_compat.h")
endif()
# The TrimUI SDK has no lto1; this path deliberately uses no release PGO.
set_property(TARGET dsperate_core PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
set(APP ${ROOT}/app/src/main/cpp/nds)
set(KOTLIN ${ROOT}/app/src/main/kotlin/com/pokedaisy/app)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(DS_DATA ${CMAKE_CURRENT_BINARY_DIR}/data_gen.h)
set(DATA_READER ${ROOT}/linux/dsperate/gen_data.py)
set(DATA_INPUTS ${KOTLIN}/GameTitles.kt)
foreach(name Gen4Reader Gen4TablesPlatinum Gen4TablesHeartGold Gen4Dex Gen4MachinesPlatinum Gen4TownMapPlatinum Gen4TownMapHeartGold PokegearMap Gen5TownMapBlack Gen5TownMapBlack2 Gen5Reader Gen5DexBlack Gen5TablesBlack Gen5ItemsBlack Gen5TablesBlack2 NdsArt RegionMapModel MonStats Gen4Charmap)
  list(APPEND DATA_INPUTS ${KOTLIN}/companion/data/${name}.kt)
endforeach()
add_custom_command(OUTPUT ${DS_DATA}
  COMMAND ${Python3_EXECUTABLE} ${DATA_READER} --output ${DS_DATA}
  DEPENDS ${DATA_READER} ${DATA_INPUTS}
  VERBATIM)
set(GUIDE_DATA ${CMAKE_CURRENT_BINARY_DIR}/guide_gen.h)
set(GUIDE_READER ${ROOT}/linux/dsperate/gen_guide.py)
set(GUIDE_INPUTS ${KOTLIN}/companion/data/GuideGen4.kt ${KOTLIN}/companion/data/GuideGen5.kt)
foreach(name GuideGen4PlatinumGen GuideGen4HeartGoldGen GuideGen5BlackGen GuideGen5WhiteGen GuideGen5Black2Gen GuideGen5White2Gen GuidePlatinum GuideHeartGold GuideBlack GuideBlack2 Gen4TownMapPlatinum Gen4TablesPlatinum Gen4TownMapHeartGold Gen5TownMapWhite Gen5TownMapWhite2 Gen5TablesBlack Gen5TablesBlack2 Gen4Reader Gen5Reader)
  list(APPEND GUIDE_INPUTS ${KOTLIN}/companion/data/${name}.kt)
endforeach()
add_custom_command(OUTPUT ${GUIDE_DATA}
  COMMAND ${Python3_EXECUTABLE} ${GUIDE_READER} ${KOTLIN}/companion/data ${GUIDE_DATA}
  DEPENDS ${GUIDE_READER} ${GUIDE_INPUTS}
  VERBATIM)
add_custom_target(ds-app-data DEPENDS ${DS_DATA} ${GUIDE_DATA})
file(GLOB PD_CORE CONFIGURE_DEPENDS ${ROOT}/3ds/core/*.c)
add_library(pd_ds_presentation STATIC companion_ui.cpp layout.cpp battle_strip.cpp battle_style.cpp ${PD_CORE}
  ${ROOT}/linux/hud.c
  ${ROOT}/linux/quick_menu.c
  ${ROOT}/linux/dsperate/companion.cpp
  ${ROOT}/linux/dsperate/party_view.cpp
  ${ROOT}/linux/dsperate/bag_view.cpp
  ${ROOT}/linux/dsperate/dex_view.cpp
  ${ROOT}/linux/dsperate/guide_view.cpp
  ${ROOT}/linux/dsperate/map_art.cpp
  ${ROOT}/linux/dsperate/map_view.cpp
  ${APP}/pk_nds_field.cpp)
add_dependencies(pd_ds_presentation ds-app-data)
target_include_directories(pd_ds_presentation PUBLIC ${CMAKE_CURRENT_BINARY_DIR} ${LR}/include
  ${ROOT}/3ds/core ${ROOT}/linux/dsperate ${APP})
target_link_libraries(pd_ds_presentation PUBLIC m)

add_library(pokedaisy_ds_libretro SHARED pd_ds_libretro.cpp ${APP}/pk_nds_call.cpp)
set_target_properties(pokedaisy_ds_libretro PROPERTIES PREFIX "")
if(WIN32)
  target_include_directories(pokedaisy_ds_libretro BEFORE PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}/windows_compat/include ${CMAKE_CURRENT_SOURCE_DIR}/windows_compat/include)
endif()
target_include_directories(pokedaisy_ds_libretro PRIVATE ${LR} ${DS}/src)
target_link_libraries(pokedaisy_ds_libretro PRIVATE pd_ds_presentation dsperate_core pthread dl)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(pokedaisy_ds_libretro PRIVATE stdc++fs)
endif()
target_link_options(pokedaisy_ds_libretro PRIVATE -Wl,--version-script=${LR}/src/link.T -Wl,--no-undefined)

# A self-contained pak should not depend on firmware C++ runtime versions.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(pokedaisy_ds_libretro PRIVATE -static-libstdc++ -static-libgcc)
endif()

    target_link_libraries(ds-settings-test PRIVATE stdc++fs)
  endif()
  add_test(NAME ds-settings COMMAND ds-settings-test ${ROOT}/app/src/test/resources/fixtures/platinum/mainram.bin ${CMAKE_CURRENT_BINARY_DIR}/validation)
  add_test(NAME ds-companion-fixture COMMAND ds-companion-fixture ${ROOT}/app/src/test/resources/fixtures/platinum/mainram.bin)
  set(DS_FIXTURES ${ROOT}/app/src/test/resources/fixtures)
  add_test(NAME ds-companion-parity COMMAND ds-companion-parity
    ${DS_FIXTURES}/heartgold/mainram.bin ${DS_FIXTURES}/soulsilver/mainram.bin ${DS_FIXTURES}/black/mainram.bin
    ${DS_FIXTURES}/white/mainram.bin ${DS_FIXTURES}/black2/mainram.bin ${DS_FIXTURES}/white2/mainram.bin)
  if(WIN32)
    add_executable(ds-windows-mman-test test/windows_mman.cpp windows_compat/windows_mman.cpp)
    target_include_directories(ds-windows-mman-test PRIVATE ${DS_WINDOWS_COMPAT}/include)
    target_compile_options(ds-windows-mman-test PRIVATE -UNDEBUG)
    add_test(NAME ds-windows-mman COMMAND ds-windows-mman-test)

endif()
